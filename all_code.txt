import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';

import 'screens/gallery_screen.dart';

final _router = GoRouter(
  routes: [
    GoRoute(path: '/',        builder: (_, __) => const TitleScreen()),
    GoRoute(path: '/play',    builder: (_, __) => const PlayScreen()),
    GoRoute(path: '/result',  builder: (_, __) => const ResultScreen()),
    GoRoute(path: '/howto',   builder: (_, __) => const HowToScreen()),
    GoRoute(path: '/score',   builder: (_, __) => const HighScoreScreen()),
    GoRoute(path: '/privacy', builder: (_, __) => const PrivacyScreen()),
    GoRoute(path: '/gallery', builder: (_, __) => const GalleryScreen()),
  ],
);

class App extends StatelessWidget {
  const App({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp.router(
      title: '競技かるた練習',
      routerConfig: _router,
      builder: (context, child) {
        final mq = MediaQuery.of(context);
        final clamped = mq.textScaleFactor.clamp(1.0, 1.3);
        return MediaQuery(
          data: mq.copyWith(textScaleFactor: clamped),
          child: child ?? const SizedBox(),
        );
      },
      theme: ThemeData(useMaterial3: true),
    );
  }
}

class RewardCatalog {
  // 未取得枠
  static const String hatena = 'assets/images/gallery/hatena.png';

  // コンプリート画像
  static const String completeImage = 'assets/images/gallery/complete.png';

  /// ギャラリーの固定スロット（インデックスが位置を表す）
  /// 例: Sが2枚, Aが2枚, ... Dが2枚, 最後がコンプリート
  static const List<String> slots = [
    'assets/images/gallery/S_01.png', // 0
    'assets/images/gallery/S_02.png', // 1
    'assets/images/gallery/A_01.png', // 2
    'assets/images/gallery/A_02.png', // 3
    'assets/images/gallery/B_01.png', // 4
    'assets/images/gallery/B_02.png', // 5
    'assets/images/gallery/C_01.png', // 6
    'assets/images/gallery/C_02.png', // 7
    'assets/images/gallery/D_01.png', // 8
    'assets/images/gallery/D_02.png', // 9
    completeImage,                    // 10 = コンプリート枠
  ];

  static const int completeIndex = 10;

  // ランク→スロット位置
  static const Map<Rank, List<int>> byRank = {
    Rank.S: [0, 1],
    Rank.A: [2, 3],
    Rank.B: [4, 5],
    Rank.C: [6, 7],
    Rank.D: [8, 9],
  };

  // 旧コードが参照しているシンプルな別名
  static List<int> get sIndices => byRank[Rank.S]!;
  static List<int> get aIndices => byRank[Rank.A]!;
  static List<int> get bIndices => byRank[Rank.B]!;
  static List<int> get cIndices => byRank[Rank.C]!;
  static List<int> get dIndices => byRank[Rank.D]!;

  // コンプリート条件: 全ランク画像が揃っている
  static bool isComplete(Set<int> owned) {
    for (final group in byRank.values) {
      for (final i in group) {
        if (!owned.contains(i)) return false;
      }
    }
    return true;
  }

  // index -> 表示名（S-1 / A-2 / ... / コンプリート）
  static String labelOf(int index) {
    if (index == completeIndex) return 'コンプリート';
    for (final entry in byRank.entries) {
      final pos = entry.value.indexOf(index);
      if (pos != -1) {
        final n = pos + 1;
        return '${entry.key.name}-$n'; // 例: S-1, D-2
      }
    }
    return '#${index + 1}';
  }

  // 通常スロット(= コンプリート以外)
  static List<int> get normalIndices =>
      List<int>.generate(slots.length - 1, (i) => i);
}

enum Rank { S, A, B, C, D }

Rank calcRank(int accuracy, double avgSec) {
  // 速いほど上：avgSec は小さいほど良い
  if (accuracy == 100 && avgSec <= 0.5) return Rank.S;
  if (accuracy >= 95 && avgSec <= 1.0) return Rank.A;
  if (accuracy >= 90 && avgSec <= 2.0) return Rank.B;
  if (accuracy >= 80 && avgSec <= 3.0) return Rank.C;
  return Rank.D;
}

class GameState extends ChangeNotifier {
  final _rng = Random();
  bool finished = false;           // ゲーム終了フラグ

  // 公開プロパティ
  late List<CardData> selfCards, enemyCards;
  late CardData current;
  int score = 0;
  int remainingTime = 10;            // UIに残秒を出したい場合用
  int elapsedSec    = 0;             // 読みはじめからの経過秒（UI表示用）

  static const int maxRound = 10;

  int round = 0;               // 何問目か (1〜10)
  int correct = 0;             // 正解数
  int wrong   = 0;             // 誤タップ数 ※タイムアップは wrong に含めない
  int totalSec = 0;            // 正解に要した秒数合計 (10 秒 - remainingTime)

  // 内部
  late List<CardData> _deck;          // ← _remaining をリネーム
  Timer? _phaseTimer;                 // 上下句 10 秒用
  DateTime _startAt = DateTime.now();

  bool _started = false;
  GameState() {
    _prepareDeck();        // 山札だけ用意、タイマーは動かさない
  }

  void startGame() {
    if (_started) return;
    _started = true;
    finished    = false;
    score       = round = correct = wrong = totalSec = 0;
    elapsedSec  = 0;

    _prepareDeck();
    _nextRound();          // ← ここで初めてタイマーが動く
  }

  void _prepareDeck() {
    _deck = List<CardData>.from(cards)..shuffle(_rng);
  }

  /* 6枚場札を作って current を決める */
  void _nextRound() {
    if (_deck.length < 6) {
      finished = true;                // デッキ不足＝終了
      _phaseTimer?.cancel();
      notifyListeners();
      return;
    }

    current = _deck.removeLast();
    _deck.shuffle(_rng);
    final wrong5 = _deck.take(5).toList();
    _deck.removeRange(0, 5);

    final List<CardData> pool = [...wrong5, current]..shuffle(_rng);
    selfCards  = pool.sublist(0, 3);
    enemyCards = pool.sublist(3);

    notifyListeners();
  }

  String rollingMsg = '';     // 表示するメッセージ
  int countdown = 0;          // 3,2,1,0 で 0 の時に流れ開始

  /// true=正解, false=誤答
  bool tap(CardData tapped) {
    final tookSec = 10 - remainingTime;
    final isCorrect = tapped == current;

    if (isCorrect) {
      correct++;
      totalSec += tookSec;
    } else {
      wrong++;
    }

    _phaseTimer?.cancel();
    _goNextSong();            // 次問題へ
    return isCorrect;
  }

  void _startPhaseTimer() {
    _phaseTimer?.cancel();
    remainingTime = 10;
    elapsedSec    = 0;
    _startAt = DateTime.now();

    _phaseTimer = Timer.periodic(const Duration(seconds: 1), (t) {
      elapsedSec    = DateTime.now().difference(_startAt).inSeconds;
      if (elapsedSec < 0) elapsedSec = 0;
      remainingTime = 10 - elapsedSec;
      if (remainingTime <= 0) {
        t.cancel();
        _handleTimeUp();
      }
      notifyListeners();
    });
  }

  void _handleTimeUp() {
    wrong++;                  // お手付き扱い
    _goNextSong();
  }

  void _goNextSong() {
    round++;                              // 進捗更新
    if (round >= maxRound) {
      finished = true;                 // 10 問終了
      _phaseTimer?.cancel();
      _saveHighScore();
      notifyListeners();
      return;
    }

    selfCards.remove(current);
    enemyCards.remove(current);
    if (_deck.isNotEmpty) {
      (selfCards.length < 3 ? selfCards : enemyCards)
          .add(_deck.removeLast());
    }

    // 新しい並べ替え（自陣⇔敵陣も再シャッフルでOK）
    final List<CardData> pool = [...selfCards, ...enemyCards]..shuffle(_rng);
    selfCards  = pool.sublist(0, 3);
    enemyCards = pool.sublist(3);

    current = pool[_rng.nextInt(6)];
    notifyListeners();
  }

  /// 3,2,1 のカウントが 0 になった瞬間に呼び出す
  void startMeasurement() {
    _startPhaseTimer();
  }

  Future<void> _saveHighScore() async {
    final prefs = await SharedPreferences.getInstance();
    final accuracy = (correct / maxRound * 100).round();
    final avgSec   = correct == 0 ? 0.0 : totalSec / correct;
    final rank     = calcRank(accuracy, avgSec).name;
    // 例： accuracy と avgSec も保存しておく
    await prefs.setInt ('bestAccuracy', accuracy);
    await prefs.setDouble('bestAvgSec', avgSec.toDouble());
    await prefs.setString('bestRank', rank);
  }

  void resetGame() {
    _phaseTimer?.cancel();
    _started = false;
    finished = false;
    score = round = correct = wrong = totalSec = 0;
    elapsedSec = 0;
    _prepareDeck();        // 次回 startGame まで待機
  }

  @override
  void dispose() {
    _phaseTimer?.cancel();
    super.dispose();
  }
}

    final ok = g.tap(c);
    _flashCenter(ctx, ok: ok);      // ★ SnackBar の代わりに中央フラッシュ
  }

  void _flashCenter(BuildContext context, {required bool ok}) {
    final overlay = Overlay.of(context, rootOverlay: true);
    if (overlay == null) return;

    final Color color = ok ? Colors.green : Colors.red;
    final String label = ok ? '正解！' : '残念';
    // ハプティックは任意
    ok ? HapticFeedback.mediumImpact() : HapticFeedback.heavyImpact();

    late OverlayEntry entry;
    entry = OverlayEntry(
      builder: (_) => Positioned.fill(
        child: IgnorePointer( // ★背面のタップをブロックしない
          child: Center(
            child: ClipRRect(
              borderRadius: BorderRadius.circular(16),
              child: BackdropFilter(
                filter: ImageFilter.blur(sigmaX: 3, sigmaY: 3),
                child: AnimatedScale(
                  duration: const Duration(milliseconds: 180),
                  curve: Curves.easeOutBack,
                  scale: 1.0,
                  child: Container(
                    padding: const EdgeInsets.symmetric(horizontal: 24, vertical: 16),
                    decoration: BoxDecoration(
                      color: color.withOpacity(0.92),
                      borderRadius: BorderRadius.circular(16),
                      boxShadow: [BoxShadow(color: color.withOpacity(0.35), blurRadius: 20)],
                    ),
                    child: Row(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        Icon(ok ? Icons.check_circle : Icons.close_rounded,
                            color: Colors.white, size: 28),
                        const SizedBox(width: 12),
                        Text(label,
                            style: const TextStyle(
                                color: Colors.white, fontSize: 22, fontWeight: FontWeight.bold)),
                      ],
                    ),
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );

    overlay.insert(entry);
    // 650ms後に自動で消す
    Future.delayed(const Duration(milliseconds: 650)).then((_) {
      // removeは1回だけ
      try { entry.remove(); } catch (_) {}
    });
  }

  Future<void> _confirmBack(BuildContext ctx, WidgetRef ref) async {
    final ok = await showDialog<bool>(
      context: ctx,
      builder: (_) => AlertDialog(
        content: const Text('タイトルへ戻りますか？'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('キャンセル')),
          TextButton(onPressed: () => Navigator.pop(ctx, true),  child: const Text('OK')),
        ],
      ),
    );
    if (ok ?? false) {
      ref.read(gameProvider).resetGame();
      if (ctx.mounted) ctx.go('/');
    }
  }

  Widget _ticker(bool counting, int counter, Animation<Offset> slide, String text, {required bool flowOn}) {
    return SizedBox(
      height: 64,                        // 高さ固定でレイアウトを揺らさない
      child: counting
          ? Center(
        child: AutoSizeText(
          '$counter',
          maxLines: 1,
          minFontSize: 20,
          style: const TextStyle(fontSize: 40, fontWeight: FontWeight.bold),
        ),
      )
          : (flowOn
          // ON: スライド表示
          ? SlideTransition(
          position: slide,
          child: AutoSizeText(
              text,
              maxLines: 2,
              minFontSize: 12,
              overflow: TextOverflow.fade,
              style: const TextStyle(fontSize: 20),
          ),
      )
    // OFF: 文字そのものを非表示に（高さは SizedBox で確保）
          : const SizedBox.shrink()),
    );
  }

  /* ------------------------------ build ------------------------------ */
  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final g        = ref.watch(gameProvider);
    final notifier = ref.read(gameProvider);
    final soundOn  = ref.watch(soundOnProvider);
    final flowOn   = ref.watch(textFlowOnProvider);
    final audio    = useMemoized(() => KarutaAudio(), []);
    useEffect(() { return () { audio.dispose(); }; }, [audio]); // 破棄

    /* ゲーム終了時に結果画面へ */
    useEffect(() {
      if (g.finished) {
        WidgetsBinding.instance.addPostFrameCallback(
              (_) => context.mounted ? context.go('/result') : null,
        );
      }
      return null;
    }, [g.finished]);

    /* カウントダウン用 hooks */
    final isCounting = useState(true);
    final counter    = useState(3);
    final animCtrl   = useAnimationController(duration: const Duration(seconds: 8));
    final slideAnim  = Tween(begin: const Offset(1, 0), end: const Offset(-1, 0))
        .animate(CurvedAnimation(parent: animCtrl, curve: Curves.linear));

    /* 3→2→1 → 詠み上げ開始（音声はプリロードしておき、0の瞬間に resume） */
    useEffect(() {
      if (isCounting.value) {
        var cancelled = false;
        Future.doWhile(() async {
          await Future.delayed(const Duration(seconds: 1));
          if (cancelled) return false;
          counter.value -= 1;
          if (counter.value == 0) {
            isCounting.value = false;
            // 同期性を上げるため同フレームで実行
            animCtrl.forward();
            // 準備済みをそのまま再生（ミュートなら音量0で進行）
            audio.playPrepared(muted: !soundOn);
            // ★ ここで 10 秒計測を開始（3,2,1 の後にスタート）
            ref.read(gameProvider).startMeasurement();
            return false;
          }
          return true;
        });
        return () { cancelled = true; };
      }
      return null;
    }, [isCounting.value, g.current.id]); // soundOnは入れない（ズレ回避）

    /* 新しい札になったらリセット */
    useEffect(() {
      // 新しい札の音源を先にロード→巻き戻して一時停止
      audio.prepareUpper(g.current.id);
      isCounting.value = false;
      Future.microtask(() => isCounting.value = true);
      counter.value = 3;
      animCtrl.reset();

// lib/services/ad_service.dart
import 'dart:io' show Platform;
import 'package:flutter/foundation.dart';
import 'package:google_mobile_ads/google_mobile_ads.dart';

class AdService {
  static bool get _androidMobile => Platform.isAndroid && !kIsWeb;

  // Google配布のテストユニットID（Android）
  static String get bannerId        => 'ca-app-pub-3940256099942544/6300978111';
  static String get interstitialId  => 'ca-app-pub-3940256099942544/1033173712';

  // ── Banner ──
  static BannerAd? createBanner() {
    if (!_androidMobile) return null;
    final ad = BannerAd(
      size: AdSize.banner,
      adUnitId: bannerId,
      request: const AdRequest(),
      listener: BannerAdListener(
        onAdFailedToLoad: (ad, err) { ad.dispose(); },
      ),
    )..load();
    return ad;
  }

  // ── Interstitial（先読み→1回表示） ──
  static InterstitialAd? _cached;

  static Future<void> prefetchInterstitial() async {
    if (!_androidMobile || _cached != null) return;
    await InterstitialAd.load(
      adUnitId: interstitialId,
      request: const AdRequest(),
      adLoadCallback: InterstitialAdLoadCallback(
        onAdLoaded: (ad) => _cached = ad,
        onAdFailedToLoad: (_) {},
      ),
    );
  }

  static Future<void> showInterstitialOnce() async {
    final ad = _cached;
    _cached = null;
    if (ad == null) return;
    await ad.show();
    ad.dispose();
    // 次回のために再度先読み（fire-and-forget）
    // unawaited(prefetchInterstitial()); を使わずに呼び出すだけでOK
    prefetchInterstitial();
  }
}

// lib/services/audio_service.dart
import 'package:audioplayers/audioplayers.dart';

class KarutaAudio {
  final AudioPlayer _player = AudioPlayer();
  int? _preparedId; // どの札をロード済みか

  KarutaAudio() {
    // 1トラックずつ再生して終わったら止まる挙動
    _player.setReleaseMode(ReleaseMode.stop);
  }

  /// 事前ロードして一時停止。0の瞬間に resume で即再生できる
  Future<void> prepareUpper(int id) async {
    final file = 'audio/${id.toString().padLeft(3, '0')}.mp3';
    if (_preparedId == id) {
      await _player.seek(Duration.zero);
      await _player.pause();
      return;
    }
    _preparedId = id;
    await _player.stop();                      // 旧トラック停止
    // バージョンにより setSource(AssetSource(...)) か setSourceAsset(...) を使います
    try {
      await _player.setSource(AssetSource(file));
    } catch (_) {
      await _player.setSourceAsset(file);      // 古いバージョン互換
    }
    await _player.seek(Duration.zero);
    await _player.pause();                     // 準備完了（無音で待機）
  }

  /// 準備済みを即再生。muted=true ならボリューム0で再生し続ける
  Future<void> playPrepared({required bool muted}) async {
    await _player.setVolume(muted ? 0.0 : 1.0);
    await _player.resume();
  }

  /// ミュート切替：進行は止めずにボリュームだけ切る/戻す
  Future<void> setMuted(bool muted) => _player.setVolume(muted ? 0.0 : 1.0);

  Future<void> stop() async {
    _preparedId = null;
    await _player.stop();
  }

  Future<void> dispose() => _player.dispose();
}

// lib/services/gallery_service.dart
import 'dart:math';
import 'package:shared_preferences/shared_preferences.dart';
import '../data/reward_catalog.dart';
import '../providers/game_state.dart';

class GalleryService {
  static const _ownedKey = 'galleryOwnedIds';
  static final _rng = Random();

  static Future<Set<int>> loadOwned() async {
    final p = await SharedPreferences.getInstance();
    final list = p.getStringList(_ownedKey) ?? const [];
    return list.map(int.parse).toSet();
  }

  static Future<void> _saveOwned(Set<int> owned) async {
    final p = await SharedPreferences.getInstance();
    await p.setStringList(_ownedKey, owned.map((e) => e.toString()).toList());
  }

  /// 1回の結果につき1枚付与:
  ///  1) まず該当ランクの未所持からランダム
  ///  2) そのランクがコンプ済みなら、他ランクの未所持からランダム
  ///  3) すべて集まった瞬間に completeIndex を自動解禁（別枠）
  static Future<int?> grantOne({required Rank rank}) async {
    final owned = await loadOwned();

    // コンプリート枠は通常の候補に含めない
    int? pickFrom(List<int> pool) {
      final remains = pool.where((i) => !owned.contains(i)).toList();
      if (remains.isEmpty) return null;
      return remains[_rng.nextInt(remains.length)];
    }

    // ランク限定：そのランクの未取得が無ければ今回は付与なし
    final group = RewardCatalog.byRank[rank]!;
    int? granted = pickFrom(group);

    // 何も残っていなければ null（以降は complete 解禁チェックのみ）
    if (granted != null) {
      owned.add(granted);
    }

    // 全ランク画像コンプ→コンプリート枠を自動解禁
    if (!owned.contains(RewardCatalog.completeIndex) &&
        RewardCatalog.isComplete(owned)) {
      owned.add(RewardCatalog.completeIndex);
      // ここでは granted は変えない（演出は結果画面の justUnlocked で別表示）
    }

    await _saveOwned(owned);
    return granted;
  }
}

import 'dart:io';
import 'dart:typed_data';
import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart' show rootBundle;
import 'package:gal/gal.dart';
import 'package:path_provider/path_provider.dart';
import 'package:share_plus/share_plus.dart';

class MediaService {
  static Future<Uint8List> _loadAssetBytes(String assetPath) async {
    final bd = await rootBundle.load(assetPath);
    return bd.buffer.asUint8List();
  }

  static String _basename(String path) => path.split('/').last;

  /// 端末ギャラリーへ保存（アルバム名はお好みで）
  static Future<bool> saveAssetToGallery(String assetPath, {String? name}) async {
    try {
      if (kIsWeb) return false;
      final bytes = await _loadAssetBytes(assetPath);
      await Gal.putImageBytes(
        bytes,
        album: '競技かるた練習',
        name: (name ?? _basename(assetPath)).replaceAll('.png', ''),
      );
      return true;
    } catch (_) {
      return false;
    }
  }

  /// 共有（X / LINE など）
  static Future<void> shareAsset(String assetPath, {String? fileName, String? text}) async {
    final bytes = await _loadAssetBytes(assetPath);
    final dir = await getTemporaryDirectory();
    final name = fileName ?? _basename(assetPath);
    final f = File('${dir.path}/$name');
    await f.writeAsBytes(bytes, flush: true);
    await Share.shareXFiles([XFile(f.path, mimeType: 'image/png', name: name)], text: text);
  }
}
